package com.hospital.management;


import java.util.Scanner;


interface manageable{
    void admitpatient();
    void dischargepatient();
}

abstract class person{
    String name;
    int age;

    person(String name, int age) throws Exception{
        if(name.isEmpty()){
            throw new Exception("Name should not be blank");
        }
        if(age<0){
            throw new Exception("Invalid Age !");
        }

        this.name = name;
        this.age = age;
    }

    abstract void displayDetails(); 
}

class Patient extends person implements manageable{
    String hospitalName;
    
    String Disease;
    static int room=10;
    final int consultationfee=500;

    Patient(String name, int age, String hospitalName, String Disease) throws Exception{
        super(name,age);
        this.hospitalName =hospitalName;
        this.Disease=Disease;
    }
    public void displayDetails(){
        System.out.println("Patient name: "+name);
        System.out.println("Patient age: "+age);
        System.out.println("Diesease: "+Disease);
    }

    public void admitpatient(){
        try{
            if(room<=0){
                throw new Exception("Rooms are full");
            }
            room--;
            System.out.println("Pateint Admited!");
            displayDetails();

            System.out.println("Hospiatal Name: "+hospitalName);
            System.out.println("Fees: "+consultationfee);
        }
        catch (Exception e){
            System.out.println("Admission Failed!"+e.getMessage());
        }
    }
    public void dischargepatient(){
        room++;
        System.out.println("Patient Discharged!");
    }

    void displayPatient(){
        System.out.println("Patient name: "+name);
        System.out.println("Patient age: "+age);
        System.out.println("Diesease: "+Disease);
        System.out.println("Hospiatal Name: "+hospitalName);
        System.out.println("Fees: "+consultationfee);
    }
}

public class hospitalmanagenmentsystem {
    public static void main(String args[]){
        Scanner sc = new Scanner(System.in);

        Patient patients[]= new Patient[5];

        int count=0;
        int choice;

        do{
            System.out.println("Hospital Management System\n");
            System.out.println("1.Admit Patient");
            System.out.println("2.Discharge Patient");
            System.out.println("3.Display Patient");
            System.out.println("4.Exit");
            
            System.out.print("Enter your choice: ");
            choice = sc.nextInt();

            sc.nextLine();

            switch(choice){
                case 1:
                    try{

                    if(count>=patients.length){
                        System.out.println("Hospital is full");
                        break;
                    }
                    
                        System.out.print("Enter patient name: ");
                        String name = sc.nextLine();
                        System.out.print("Enter patient age: ");
                        int age = sc.nextInt();
                        sc.nextLine();
                        System.out.print("Enter Hospital name: ");
                        String hospitalname = sc.nextLine();
                        System.out.print("Enter Disease name: ");
                        String Disease = sc.nextLine();

                        patients[count]= new Patient(name,age,hospitalname,Disease);
                        patients[count].admitpatient();
                        count++;
                    
                    }
                    catch (Exception e){
                        System.out.println("Error "+e.getMessage());
                    }
                    break;
                case 2:
                    if(count==0){
                        System.out.println("No patient found!");
                    }
                    else{
                        System.out.print("Enter patient name for discharge: ");
                        String Discharge = sc.nextLine();

                        boolean found = false;

                        for(int i=0;i<count;i++){
                            if(patients[i]!=null && patients[i].name.equalsIgnoreCase(Discharge)){
                                patients[i].dischargepatient();

                                for(int j=i;j<count-1;j++){
                                    patients[j]=patients[j+1];
                                }
                                patients[count-1]=null;
                                count--;
                                found = true;
                                System.out.println("Patient dischaged!");
                                break;
                            }
                        }
                        if(!found){
                            System.out.println("Patient not found!");
                        }
                    }
                    break;
                case 3:
                    if(count==0){
                        System.out.println("No patient");
                    }   
                    else{
                        System.out.println("Patient Record\n");
                        for(int i=0;i<count;i++){
                            patients[i].displayPatient();
                        }
                    }
                    break;
                case 4:
                    System.out.println("Thanks for visiting");
                    break;

                default :
                    System.out.println("Invalid input!");

            }
        }while(choice!=4);
        sc.close();

    }
}
